+]
+
+
+class Solution:
+    def twoSum(self, numbers: list[int], target: int) -> list[int]:
+        """Find 1-indexed positions of two numbers summing to target.
+
+        Uses two-pointer inward convergence on the pre-sorted array to locate
+        the unique pair in O(n) time and O(1) auxiliary space.
+
+        Args:
+            numbers: 1-indexed list of integers sorted in non-decreasing order.
+            target: Target sum.
+
+        Returns:
+            List containing the two 1-based indices [index1, index2].
+
+        Raises:
+            ValueError: If no pair sums to target (violating problem constraints).
+        """
+        # Initialize two pointers at opposite ends of the sorted array.
+        # Starting at the extrema allows deterministic elimination of impossible values.
+        left = 0
+        right = len(numbers) - 1
+
+        # Inward convergence loop searching for target sum.
+        # Since exactly one solution is guaranteed, the pointers will meet the solution before crossing.
+        while left < right:
+            current_sum = numbers[left] + numbers[right]
+
+            # Matching pair found: convert 0-based indices to 1-based indexing as required by problem.
+            if current_sum == target:
+                return [left + 1, right + 1]
+
+            # Current sum is deficient: advance left pointer to larger value to increase sum.
+            if current_sum < target:
+                left += 1
+            # Current sum is excessive: retreat right pointer to smaller value to decrease sum.
+            else:
+                right -= 1
+
+        # Defensive guard clause triggered only if inputs violate the guarantee of exactly one solution.
+        raise ValueError("No two sum solution exists for given target")

+class Solution:
+    def threeSum(self, nums: list[int]) -> list[list[int]]:
+        """Find all unique triplets summing to zero.
+
+        Sorts the array and iterates each unique candidate element as the first value,
+        using two-pointer inward convergence on the remaining suffix to locate pairs
+        summing to -nums[i]. Achieves O(n^2) time and O(1) auxiliary space (excluding output).
+
+        Args:
+            nums: List of integers.
+
+        Returns:
+            List of unique triplets [a, b, c] such that a + b + c == 0.
+        """
+        # Sort in non-decreasing order to enable two-pointer scanning and duplicate elimination.
+        # In-place sorting operates in O(n log n) time and O(1) auxiliary space in Python (Timsort).
+        nums.sort()
+        triplets: list[list[int]] = []
+        n = len(nums)
+
+        # Iterate through candidate first elements.
+        # Stop at n - 2 since a valid triplet requires at least three elements.
+        for i in range(n - 2):
+            # Guard clause: if current value exceeds zero, no subsequent non-negative numbers
+            # can sum to zero. Early break prunes impossible search branches.
+            if nums[i] > 0:
+                break
+
+            # Skip duplicate values for the first element to avoid redundant triplets.
+            # Comparison with nums[i - 1] ensures we only process the first occurrence.
+            if i > 0 and nums[i] == nums[i - 1]:
+                continue
+
+            left = i + 1
+            right = n - 1
+
+            # Two-pointer scan on the remaining sorted sub-array.
+            while left < right:
+                total = nums[i] + nums[left] + nums[right]
+
+                # Target triplet found: record and advance both pointers past duplicate values.
+                if total == 0:
+                    triplets.append([nums[i], nums[left], nums[right]])
+
+                    # Advance left pointer past all consecutive identical values.
+                    while left < right and nums[left] == nums[left + 1]:
+                        left += 1
+                    # Decrement right pointer past all consecutive identical values.
+                    while left < right and nums[right] == nums[right - 1]:
+                        right -= 1
+
+                    left += 1
+                    right -= 1
+
+                # Sum is too small: increment left pointer to increase total.
+                elif total < 0:
+                    left += 1
+                # Sum is too large: decrement right pointer to decrease total.
+                else:
+                    right -= 1
+
+        return triplets
